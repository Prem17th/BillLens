-- Query: Detect 30-Day Purchase Price Creep on High-Velocity SKUs
SELECT 
    i.sku_code,
    i.product_name,
    s.supplier_name,
    h.historical_rate AS prev_rate_inr,
    i.billed_rate AS current_rate_inr,
    ROUND(((i.billed_rate - h.historical_rate) / h.historical_rate) * 100, 2) AS price_creep_pct,
    (i.billed_rate - h.historical_rate) * i.quantity AS monthly_leakage_inr
FROM aurora_invoice_items i
JOIN aurora_suppliers s ON i.supplier_id = s.supplier_id
JOIN aurora_sku_price_history h ON i.sku_code = h.sku_code
WHERE i.billed_rate > h.historical_rate
ORDER BY monthly_leakage_inr DESC;
